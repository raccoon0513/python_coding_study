# Q1. 소용돌이 수
## 문제
입력으로 $n$이 주어지고,
$n \times n$ 사이즈의 소용돌이 행렬이 만들어진다.
```
n = 3
1 2 3
8 9 4
7 6 5
```
1부터 시작하여 대각선에 위치한 값들의 합을 리턴하시오.

## 예시
n=1
```
1
```
n=2
```
4
```
n=3
```
15
```

```python
# Q1. 소용돌이 수
# -*- coding: utf-8 -*-
# UTF-8 encoding when using korean
def solution(n):
    arr = [[0]*n for _ in range(n)]
    x,y= 0,0
    route_axis = 0 # route_axis % 4
    route = [[1, 0], [0,1],[-1,0], [0,-1]]
    for i in range(1, n**2):
        arr[x][y] = i
        nx, ny= x + route[route_axis%4][0], y + route[route_axis%4][1]
        if not ((0 <= nx < n) and (0 <= ny < n) and (arr[nx][ny]==0)):
            route_axis+=1
            nx, ny= x + route[route_axis%4][0], y + route[route_axis%4][1]
        x, y = nx, ny
    arr[x][y] = (n**2)
    return sum([arr[i][i] for i in range(n)])

n1=3
ret1 = solution(n1)
print(f"solution 함수의 반환 값은 {ret1}입니다")

n2=2
ret1 = solution(n2)
print(f"solution 함수의 반환 값은 {ret1}입니다")
```

# Q2. 체스의 나이트
## 문제
8x8 체스판 위에 존재하는 나이트의 위치가 입력될 떄,

나이트가 당장 이동 가능한 칸 개수를 구하시오

## 예시
```
A7
3
```

```python
# Q2. 체스의 나이트
# -*- coding: utf-8 -*-
# UTF-8 encoding when using korean
def solution(pos):
    ascii_arr = "ABCDEDFGH"
    x = ascii_arr.index(pos[0])
    y = int(pos[1]) - 1
    count = 0
    # 반시계 회전 
    moveset_x = [-1, -2, -2 , -1, 1, 2, 2, 1]
    moveset_y = [2, 1, -1, -2, -2, -1, 1, 2]
    for i in range(8):
        if (0<=x+moveset_x[i]<8) and (0<=y+moveset_y[i]<8) : 
            count += 1 
    return count

pos = "A7"
ret = solution(pos)
try:
    print(f"ret : {ret}")
except Exception as e:
    pass
```

# Q3. 타임머신

구진법 연산을 해내어보시오.

구지법을 사용하는 자연수 n이 주어지고, n+1을 리턴하면 됩니다.
```python
# Q3. 타임머신
# -*- coding: utf-8 -*-
# UTF-8 encoding when using korean
def solution(num):
	temp_num = num
	answer = 0
	num_arr = [] # 역순으로 저장됨
	while(temp_num > 0):
		num_arr.append(temp_num%10)
		temp_num = temp_num//10
	upper_flag = True
	for i in range(len(num_arr)):
		if upper_flag:
			num_arr[i] += 1
			if num_arr[i] == 10:
				num_arr[i] = 1
				upper_flag = True	
			else :
				upper_flag = False
	if upper_flag:
		num_arr.append(1)
	ten_key = 1
	for i in num_arr:
		answer+= i * ten_key
		ten_key*=10
	return answer
num = 9949999
ret = solution(num)
 
print("solution 함수의 반환 값은", ret, "입니다.")
```
